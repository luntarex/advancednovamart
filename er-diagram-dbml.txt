Table users {
  id bigint [pk, increment]
  email varchar [not null, unique]
  password varchar [not null]
  role_type varchar [not null]
  gender varchar
  created_at datetime
}

Table customer_profiles {
  id bigint [pk, increment]
  user_id bigint [not null, unique]
  age int
  city varchar
  membership_type varchar
  total_spend double
  items_purchased int
  avg_rating double
  discount_applied boolean
  satisfaction_level varchar
}

Table stores {
  id bigint [pk, increment]
  name varchar [not null]
  owner_id bigint [not null]
  status varchar [not null]
  created_at datetime
}

Table categories {
  id bigint [pk, increment]
  name varchar [not null]
  parent_id bigint
}

Table products {
  id bigint [pk, increment]
  name varchar [not null]
  sku varchar [not null, unique]
  description text
  image_url varchar
  unit_price decimal [not null]
  stock_quantity int [not null]
  category_id bigint
  store_id bigint
}

Table orders {
  id bigint [pk, increment]
  user_id bigint [not null]
  store_id bigint
  status varchar [not null]
  grand_total decimal [not null]
  payment_method varchar
  order_date datetime
}

Table order_items {
  id bigint [pk, increment]
  order_id bigint [not null]
  product_id bigint [not null]
  quantity int [not null]
  price decimal [not null]
}

Table shipments {
  id bigint [pk, increment]
  order_id bigint [not null, unique]
  warehouse varchar
  mode varchar
  tracking_number varchar [unique]
  status varchar [not null]
  estimated_delivery date
  last_updated datetime
}

Table reviews {
  id bigint [pk, increment]
  user_id bigint [not null]
  product_id bigint [not null]
  star_rating int [not null]
  review_text text
  sentiment varchar
  helpful_votes int
  response_text text
  visibility varchar
  created_at datetime
}

Table addresses {
  id bigint [pk, increment]
  user_id bigint [not null]
  address_line varchar [not null]
  city varchar [not null]
  district varchar
  phone varchar
  is_default boolean [not null]
  created_at datetime
}

Ref: customer_profiles.user_id > users.id
Ref: stores.owner_id > users.id
Ref: categories.parent_id > categories.id
Ref: products.category_id > categories.id
Ref: products.store_id > stores.id
Ref: orders.user_id > users.id
Ref: orders.store_id > stores.id
Ref: order_items.order_id > orders.id
Ref: order_items.product_id > products.id
Ref: shipments.order_id > orders.id
Ref: reviews.user_id > users.id
Ref: reviews.product_id > products.id
Ref: addresses.user_id > users.id
